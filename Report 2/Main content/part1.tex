\section{EXECUTIVE SUMMARY}

\subsection{Tổng quan cơ hội \& Vấn đề giải quyết}
Phương pháp học tiếng Anh truyền thống thường dựa trên các bài tập rời rạc, dễ gây nhàm chán và thiếu môi trường thực hành thực tế. Dự án \textbf{AI English Learning RPG} ra đời nhằm thay đổi cách tiếp cận này bằng việc kết hợp giữa giáo dục và giải trí (Edutainment). Dự án hướng tới xây dựng một ứng dụng trò chơi di động nhập vai (Mobile RPG), nơi người học bắt buộc phải sử dụng tiếng Anh như một công cụ chính để sinh tồn, tương tác và hoàn thành các thử thách trong game thay vì làm bài tập một cách thụ động.

\subsection{Giải pháp \& Giá trị cốt lõi}
Trọng tâm của giải pháp là việc ứng dụng Trí tuệ nhân tạo (AI) để cá nhân hóa lộ trình học tập của từng người thông qua một vòng lặp học tập tự động (Learning Loop):

$$\text{Chơi (Play)} \rightarrow \text{Sử dụng tiếng Anh (Use English)} \rightarrow \text{Mắc lỗi (Make mistake)}$$ 
$$\rightarrow \text{AI phân tích lỗi (AI analyze)} \rightarrow \text{Tạo nhiệm vụ phù hợp (Generate quest)} \rightarrow \text{Áp dụng lại (Apply again)}$$

\vspace{0.3cm}
Sản phẩm mang lại 4 giá trị giáo dục cốt lõi cho người dùng:
\begin{itemize}
    \item \textbf{Tăng vốn từ vựng theo ngữ cảnh:} Từ vựng gắn liền với cốt truyện và tình huống trong game giúp ghi nhớ sâu.
    \item \textbf{Cải thiện từ vựng thực tế:} Học cách áp dụng từ vựng thông qua các bối cảnh cụ thể của các câu hỏi.
    \item \textbf{Tập phản xạ hội thoại:} Xây dựng tư duy phản xạ tiếng Anh tự nhiên.
    \item \textbf{Cá nhân hóa:} Hệ thống tự động nhận diện và tập trung cải thiện các điểm yếu riêng của từng người học.
\end{itemize}

\subsection{Công nghệ chủ chốt \& Kiến trúc hệ thống}
Để hiện thực hóa trải nghiệm nhập vai thông minh, hệ thống loại bỏ các tính năng như sửa lỗi câu trực tiếp, nhận diện giọng nói (Speech-to-text) hay thuật toán phát âm để tập trung tối ưu hóa vào các công nghệ AI chạy cục bộ (Local) và hệ thống dữ liệu nhúng sau:
\begin{itemize}
    \item \textbf{Trí tuệ nhân tạo (Generative AI \& LLM):} Sử dụng cổng API của Ollama để vận hành mô hình ngôn ngữ lớn mã nguồn mở Qwen 2.5 (3B) trực tiếp trên thiết bị, giúp sinh lời thoại tự động cho NPC (\textit{LLM dialogue generation}) và tự động thiết kế các nhiệm vụ học tập dựa trên khuôn mẫu có sẵn (\textit{Quest generator LLM + template}).
    \item \textbf{Thuật toán cá nhân hóa:} Ứng dụng hệ thống phân tích hồ sơ lỗi của người chơi (\textit{Player error profiling})qua độ thành thạo các từ vựng nhằm xác định điểm yếu, và bộ quy tắc điều chỉnh độ khó thích ứng (\textit{Adaptive difficulty rules}) để tự động tăng/giảm thử thách trong game và cá nhân hóa trải nghiệm ôn tập từ vựng.
    \item \textbf{Nền tảng phát triển \& Lưu trữ cục bộ:} Sử dụng Godot Engine (GDScript) làm nền tảng điều phối logic cốt lõi của game và tích hợp cơ sở dữ liệu nhúng SQLite ngay trên máy để lưu trữ tiến độ chơi, bộ sổ tay từ vựng (\textit{notebook}) cùng nhật ký lỗi từ vựng của người dùng một cách bảo mật, độc lập và tối ưu hiệu năng.
\end{itemize}

\subsection{Kết quả kỳ vọng}
AI English Learning RPG kỳ vọng sẽ tạo ra một bước đột phá trong mảng ứng dụng học ngôn ngữ, biến quá trình học tập áp lực thành một trải nghiệm giải trí lôi cuốn. Dự án không chỉ chứng minh tính khả thi về mặt kỹ thuật trong việc tích hợp trực tiếp LLM chạy local (Ollama) vào công cụ phát triển game (Godot Engine) và hệ quản trị dữ liệu nhúng (SQLite) mà còn mang lại giá trị thực tiễn cao, giúp người học duy trì động lực và phát triển phản xạ tiếng Anh một cách tự nhiên nhất.


\section{PROJECT OVERVIEW}

\subsection{Bối cảnh dự án (Project Background)}
Việc duy trì động lực khi học một ngôn ngữ mới luôn là một thách thức lớn đối với hầu hết người học. Các phương pháp học truyền thống hoặc các ứng dụng hiện tại thường đi theo lối mòn:học từ vựng qua thẻ ghi nhớ (flashcard) một cách rời rạc, thiếu đi tính ngữ cảnh và môi trường thực hành.

Dự án \textbf{AI English Learning RPG} được phát triển nhằm xóa bỏ rào cản đó. Bằng cách lồng ghép việc học vào một trò chơi nhập vai di động (Mobile RPG), dự án biến tiếng Anh từ một "môn học áp lực" thành một "công cụ sinh tồn và khám phá". Người chơi buộc phải sử dụng ngôn ngữ để tương tác với thế giới trong game, từ đó hình thành phản xạ tự nhiên và duy trì được hứng thú học tập lâu dài.

\subsection{Mục tiêu dự án (Project Objectives)}

\noindent \textbf{Về khía cạnh Học tập (Learning Objectives)}
\begin{itemize}
    \item \textbf{Tối ưu hóa từ vựng theo ngữ cảnh:} Giúp người học tiếp thu và ghi nhớ từ vựng thông qua cốt truyện, bối cảnh cốt truyện và các tình huống giao tiếp cụ thể trong game.
    \item \textbf{Ứng dụng từ vựng thực tế:} Tạo môi trường để người học vận dụng từ vựng khi tương tác, thay vì học vẹt lý thuyết.
    \item \textbf{Cá nhân hóa lộ trình:} Tự động nhận diện, phân tích và tập trung cải thiện các lỗ hổng kiến thức riêng biệt của từng người chơi.
\end{itemize}

\noindent \textbf{Về khía cạnh Kỹ thuật (Technical Objectives)}
\begin{itemize}
    \item \textbf{Tích hợp Trí tuệ nhân tạo (LLM):} Ứng dụng thành công mô hình ngôn ngữ lớn để xử lý hội thoại động và tự động hóa việc thiết kế nhiệm vụ.
    \item \textbf{Xây dựng Vòng lặp học tập tự động (Learning Loop):} Thiết lập hệ thống khép kín từ lúc người chơi hành động, mắc lỗi, AI phân tích cho đến khi tạo ra thử thách mới để người chơi sửa sai.
    \item \textbf{Tối ưu hóa hiệu năng và lưu trữ local:} Xây dựng hệ thống vận hành mượt mà theo mô hình ứng dụng độc lập (Standalone Application), quản lý dữ liệu của người học một cách toàn vẹn thông qua SQLite mà không phụ thuộc vào kết nối Internet hay máy chủ đám mây bên thứ ba.
\end{itemize}

\subsection{Phạm vi dự án (Project Scope)}
Để đảm bảo dự án hoàn thành đúng tiến độ và tập trung vào các tính năng cốt lõi mang lại giá trị cao nhất, phạm vi công việc được quy định rõ ràng như sau:

\vspace{0.2cm}
\noindent \textbf{Các tính năng nằm TRONG phạm vi phát triển (In-Scope)}
\begin{itemize}
    \item \textbf{LLM Quest Generation:} Hệ thống AI tạo câu hỏi và  phản hồi câu trả lời dựa trên ngữ cảnh game và hành vi của người chơi.
    \item \textbf{Quest Generator (LLM + Template):} Tự động tạo các nhiệm vụ học thuật trong game bằng cách kết hợp sức mạnh của LLM và các khuôn mẫu định sẵn.
    \item \textbf{Player Error Profiling:} Trình phân tích ghi nhận và thiết lập hồ sơ các lỗi sai (từ vựng) mà người chơi gặp phải.
    \item \textbf{Adaptive Difficulty Rules:} Bộ quy tắc tự động điều chỉnh, tăng hoặc giảm độ khó của game dựa trên hồ sơ năng lực của người chơi.
    \item \textbf{Vocabulary Extraction} Hệ thống trích xuất từ có độ thành thạo thấp từ những lần chơi và tổng hợp lại để hỗ trợ người chơi ôn tập.
    \item \textbf{Game Logic \& Client Controller (Godot Engine / GDScript):} Xây dựng các module xử lý logic cốt lõi của trò chơi và điều phối luồng dữ liệu.
    \item \textbf{Local Database Management (SQLite):} Lưu trữ cục bộ và quản lý toàn vẹn tiến độ chơi, sổ tay từ vựng (notebook) cần ôn tập.
\end{itemize}

\noindent \textbf{Các tính năng nằm NGOÀI phạm vi phát triển (Out-of-Scope)}
\begin{itemize}
    \item \textbf{Pronunciation Heuristic:} Không phát triển công cụ tự động phát hiện và đánh giá lỗi phát âm sai.
    \item \textbf{Grammar Correction (LLM):} Không tích hợp tính năng LLM sửa lỗi câu trực tiếp ngay trong đoạn chat hội thoại.
    \item \textbf{Speech-to-text:} Không hỗ trợ tính năng nhập liệu hoặc giao tiếp bằng giọng nói (chỉ tập trung vào text-based nhập vai).
\end{itemize}

\subsection{Khung công nghệ sử dụng (Technology Stack)}
Để đảm bảo tính độc lập, tối ưu hóa hiệu năng xử lý local trên máy tính cá nhân (PC) và không phụ thuộc vào kết nối Internet hay chi phí dịch vụ đám mây bên thứ ba, dự án được xây dựng trên một kiến trúc Local-first hoàn chỉnh với các công nghệ sau:

\begin{itemize}
    \item \textbf{Game Engine \& Core Logic Platform: Godot Engine (GDScript)} \\
    Được lựa chọn làm nền tảng phát triển tổng thể cho cả giao diện trò chơi (Client UI/UX) và bộ điều khiển logic trung tâm. GDScript đảm nhận nhiệm vụ xử lý trạng thái game, điều phối luồng dữ liệu, thực hiện các thuật toán lặp lại ngắt quãng (Spaced Repetition) và bộ quy tắc điều chỉnh độ khó thích ứng (Adaptive Difficulty Rules).
    
    \item \textbf{Local AI API Server \& Models:}
    \begin{itemize}
        \item \textbf{Ollama API:} Đóng vai trò là cổng API cục bộ (Local API Server) để quản lý và vận hành mô hình trí tuệ nhân tạo trực tiếp trên máy của người dùng. Godot sẽ giao tiếp with Ollama thông qua giao thức HTTP REST Client tiêu chuẩn (\textit{HTTPRequest node}).
        \item \textbf{Qwen 2.5 (Phiên bản 3B - 3 Billion Parameters):} Mô hình ngôn ngữ lớn (LLM) mã nguồn mở chạy local, được tối ưu riêng cho các tác vụ ngôn ngữ và hội thoại. Model này chịu trách nhiệm sinh lời thoại tự động cho NPC (\textit{LLM Dialogue Generation}) và phối hợp với các khuôn mẫu để tự động thiết kế nhiệm vụ học tập (\textit{Quest Generator}).
    \end{itemize}
    
    \item \textbf{Database Management System: SQLite} \\
    Hệ quản trị cơ sở dữ liệu quan hệ nhúng (Embedded Database) nhỏ gọn, được tích hợp trực tiếp vào trong Project Godot thông qua \textit{The godot-sqlite plugin}. Toàn bộ dữ liệu về tiến độ chơi, bộ sổ tay từ vựng cá nhân (Notebook), nhật ký lỗi sai và hồ sơ năng lực của người chơi sẽ được lưu trữ cục bộ một cách an toàn và truy xuất với tốc độ cao.
\end{itemize}


\section{PHÂN TÍCH VẤN ĐỀ (PROBLEM ANALYSIS)}

\subsection{Problem Statement (chuẩn hóa bài toán)}
Sinh viên thường mất động lực học tiếng Anh do từ vựng khó nhớ và bối cảnh khô khan. Bài toán đặt ra là xây dựng một hệ thống RPG tích hợp AI để biến quá trình học thành trò chơi giải trí, tăng hứng thú và khả năng ghi nhớ thông qua tương tác cốt truyện.
\begin{enumerate}
	\item \textbf{Đầu vào (Input):} 
	\begin{itemize}
		\item Thông số ngữ cảnh hiện tại từ Game Engine truyền xuống: Độ thành thạo từ vựng của người chơi hiện tại.
		\item Hệ khuôn mẫu câu lệnh (System Prompt Template) được thiết lập sẵn để định hình cấu trúc dữ liệu đầu ra cho AI.
	\end{itemize}
	\item \textbf{Đầu ra (Output):} 
	\begin{itemize}
		\item Một chuỗi dữ liệu văn bản đã được định dạng cấu trúc (JSON Object) chứa chính xác các trường thông tin: "question" (nội dung câu hỏi), "correct\_answer" (đáp án đúng), và "distractors" (mảng gồm 3 đáp án gây nhiễu) cũng như là "explaination" (phần giải thích) nếu người chơi trả lời sai.
	\end{itemize}
		\item \textbf{Ràng buộc (Constraints):}
	\begin{itemize}
		\item \textit{Ràng buộc kỹ thuật / Phần cứng:} Hệ thống phải chạy Offline 100\% trên máy trạm (Local), không sử dụng API Internet bên ngoài, giới hạn dung lượng bộ nhớ RAM phân bổ cho AI nhỏ hơn hoặc bằng 8GB.
	\end{itemize}
\end{enumerate}

\subsection{Requirements}
\begin{itemize}
	\item \textbf{Yêu cầu chức năng (Functional Requirements):} 
	\begin{itemize}
		\item Hệ thống phải tự động nhận biết khi người chơi kích hoạt hội thoại với NPC để gửi lệnh sinh câu hỏi.
		\item Hệ thống phải kiểm tra tính đúng/sai của đáp án do người chơi chọn và cập nhật trình độ của Player.
	\end{itemize}	 
	\item \textbf{Yêu cầu phi chức năng (Non-functional Requirements):}
	\begin{itemize}
		\item Mô hình AI Local có thời gian phản hồi thấp và hiệu suất cao, điều này cần tối ưu API để thời gian phản hồi <= 5 giây
		\item Độ chính xác của nội dung câu hỏi được AI sinh ra ổn định, tỷ lệ câu hỏi đúng ngữ pháp và cấu trúc >= 90\%
	\end{itemize}
\end{itemize}

\subsection{Constraints \& Assumptions (Ràng buộc \& Giả định)}

\subsubsection{Ràng buộc (Constraints)}
Đây là các giới hạn bắt buộc nhóm phát triển phải tuân thủ, không thể thay đổi do đặc thù của công nghệ và đề tài:
\begin{itemize}
    \item \textbf{Ràng buộc Phần cứng \& Tài nguyên (Hardware Constraints):} Hệ thống bắt buộc phải vận hành mượt mà trên các máy trạm cá nhân (PC) cấu hình phổ thông. Dung lượng bộ nhớ RAM phân bổ tối đa cho mô hình AI chạy cục bộ (Local LLM) phải thỏa mãn điều kiện $\le \text{8GB}$.
    \item \textbf{Ràng buộc Kết nối AI:} Game tuyệt đối không gọi các API từ bên thứ 3 như OpenAI (ChatGPT), Anthropic (Claude) hay Gemini. Thay vào đó, Engine game (Godot) chỉ kết nối và giao tiếp với AI thông qua cổng mạng nội bộ (Localhost) tới service Ollama đang chạy ngầm trên cùng một máy tính của người chơi. Vì không kết nối cloud, hệ thống bị ràng buộc bởi sức mạnh phần cứng của máy tính người chơi (CPU/GPU và RAM) để chạy mô hình \textbf{Qwen2.5-3B-Instruct}.
    \item \textbf{Ràng buộc về Cơ sở dữ liệu:} Godot kết nối trực tiếp với file data.db (SQLite) được đặt cùng cấp trong thư mục game. Mọi thao tác truy vấn (Query), đọc/ghi tiến trình học tập, trạng thái lưu game đều diễn ra trên ổ cứng cục bộ.
    \item \textbf{Ràng buộc về Tài khoản và Người dùng:} Người chơi không cần đăng nhập, đăng ký bằng email hay mạng xã hội. Dữ liệu (Player Profile) được lưu ở file \texttt{data.db} ngay trên máy đó. Do đó, người chơi chơi trên máy nào thì dữ liệu lưu trên máy đó. Nếu muốn chuyển sang máy khác, người dùng phải copy file \texttt{data.db} mang theo.
    \item \textbf{Ràng buộc về Phạm vi Tính năng (Scope Constraints):} Loại bỏ hoàn toàn các tính năng phức tạp bao gồm sửa lỗi câu trực tiếp từ LLM trong đoạn hội thoại (Grammar Correction), nhận diện giọng nói (Speech-to-text), và thuật toán đánh giá phát âm (Pronunciation Heuristic) để tập trung tài nguyên tối ưu cho trải nghiệm nhập vai dựa trên văn bản (Text-based RPG).
    \item \textbf{Ràng buộc về Định dạng Đầu ra (Output Formatting Constraints):} Phản hồi từ mô hình ngôn ngữ lớn chạy local phải trả về chính xác định dạng cấu trúc dữ liệu chuỗi văn bản đối tượng JSON (JSON Object) với các trường quy định rõ ràng bao gồm: \texttt{"question"}, \texttt{"correct\_answer"}, và \texttt{"distractors"} để cấu phần Game Engine có thể bóc tách dữ liệu thành công.
\end{itemize}

\subsubsection{Giả định (Assumptions)}
Đây là các yếu tố được giả định là đúng để làm cơ sở và tiền đề thiết kế hệ thống kỹ thuật:
\begin{itemize}
    \item \textbf{Giả định về năng lực Mô hình:} Giả định rằng mô hình mã nguồn mở Qwen 2.5 (phiên bản 3B Billion Parameters) khi vận hành cục bộ thông qua Ollama API có đủ khả năng hiểu ngữ cảnh trò chơi và sinh câu hỏi trắc nghiệm ổn định, đạt độ chính xác kiểm soát nội dung $\ge \text{90\%}$ thông qua các kỹ thuật thiết lập khuôn mẫu câu lệnh (Prompt Engineering).
    \item \textbf{Giả định về hành vi Người dùng:} Giả định rằng đối tượng sinh viên (người dùng cuối) sở hữu trình độ năng lực tiếng Anh nền tảng nằm trong khung từ bậc A1 đến B2, và sẽ thực hiện tương tác làm bài một cách nghiêm túc để hệ thống xây dựng hồ sơ năng lực (Player Mastery Profile) chính xác.
    \item \textbf{Giả định về hiệu năng Godot-SQLite:} Giả định plugin \texttt{godot-sqlite} hoạt động ổn định trên Godot Engine, đảm bảo việc thực thi đọc/ghi dữ liệu lịch sử lỗi sai, tiến trình chơi và hàng đợi câu hỏi diễn ra bất đồng bộ mà không gây ra hiện tượng nghẽn luồng xử lý đồ họa và logic chính của trò chơi (Main Game Loop Thread).
\end{itemize}

\subsection{Các Core Feature}


\begin{enumerate}
	\item \textbf{Feature 1: Hệ thống Sinh nội dung Thích ứng (Adaptive Content Generation \& DDA)} 
	
	\begin{itemize}
		\item \textbf{Mô tả:} Thay vì sử dụng ngân hàng câu hỏi tĩnh tĩnh, hệ thống tích hợp thuật toán \textit{Điều chỉnh độ khó động (Dynamic Difficulty Adjustment - DDA)} kết hợp với AI Prompting. Dựa trên mức độ thành thạo hiện tại của người chơi, Godot sẽ điều chỉnh prompt gửi tới Ollama để sinh ra các câu hỏi trắc nghiệm hoàn toàn phù hợp với trình độ người chơi theo thời gian thực.
		\item \textbf{Giải quyết Pain Point:} Xóa bỏ sự nhàm chán của việc lặp lại câu hỏi cố định; giữ người chơi ở trạng thái "Flow" (không quá dễ gây chán, không quá khó gây nản).
	\end{itemize}
	\item  \textbf{Feature 2: Hệ thống Đánh giá \& Phản hồi AI Tức thời (Real-time Evaluation \& Explainable AI)}
	\begin{itemize}
		\item \textbf{Mô tả:} Ngay sau mỗi lượt tương tác/chiến đấu, thay vì chỉ trả về kết quả Đúng/Sai (Binary evaluation), hệ thống sử dụng năng lực của LLM để thực hiện \textit{Phản hồi có thể giải thích (Explainable AI Feedback)}. Cụ thể, AI sẽ chỉ ra lỗi sai cốt lõi và cung cấp lời giải thích ngắn gọn, dễ hiểu, đặt chuẩn xác vào bối cảnh của câu hỏi vừa làm.
		\item \textbf{Giải quyết Pain Point:} Giúp người học hiểu sâu bản chất vấn đề thay vì học vẹt (nhớ đáp án A, B, C). Cung cấp trải nghiệm như có một "\textit{Gia sư 1-1}" đồng hành ngay trong trận chiến.
	\end{itemize}
	\item \textbf{Feature 3: Hệ thống Theo dõi Kiến thức \& Cá nhân hóa Ôn tập (Knowledge Tracing)} 
	\begin{itemize}
		\item \textbf{Mô tả:} Mô tả: Mọi tương tác của người chơi được phân tích và lưu trữ vào SQLite để tạo thành một "Hồ sơ năng lực" (Player Mastery Profile). Hệ thống áp dụng mô hình \textit{Theo dõi kiến thức (Knowledge Tracing)} để tự động nhận diện các từ vựng mới, định lượng điểm yếu từ vựng (dựa trên tần suất trả lời sai) và  tổng hợp để người chơi tự ôn tập.
		\item \textbf{Giải quyết Pain Point:} Khắc phục triệt để "đường cong quên lãng" (Ebbinghaus forgetting curve). Tối ưu hóa thời gian học bằng cách cá nhân hóa 100\% lộ trình: chỉ học những gì mình chưa biết hoặc hay làm sai.
	\end{itemize}
	\item \textbf{Feature 4: Hệ thống Sinh nội dung Theo ngữ cảnh Game (Context-Aware Thematic Prompting)}
	\begin{itemize}
		\item \textbf{Mô tả:} Thay vì sinh các câu hỏi trắc nghiệm từ vựng với nội dung ngẫu nhiên, hệ thống sẽ chèn thông số về môi trường (Biome) hoặc loại quái vật (Enemy Type) hiện tại của người chơi vào \textit{AI Prompt}. Ví dụ: Nếu người chơi đang ở "Khu rừng Lửa" (Fire Cave) và cần học từ "Vulnerable", AI (Ollama) sẽ sinh ra một câu hỏi trắc nghiệm có nội dung về hiệp sĩ, ngọn lửa, rồng hoặc sự sinh tồn.
		\item \textbf{Giải quyết Pain Point:} Người học thường cảm thấy chán nản vì phải làm các câu hỏi từ vựng khô khan, rời rạc, không liên quan đến nhau. Tính năng này giúp gắn kết chặt chẽ (Immersive) trải nghiệm học tập vào cốt truyện và bối cảnh game, tăng sự hứng thú.
	\end{itemize}
	
	\item \textbf{Feature 5: Hệ thống Phân cấp Năng lực theo Tuyến Quái vật (Enemy-bound Hierarchical Competency Gating)}
	\begin{itemize}
		\item \textbf{Mô tả:} Trong phạm vi một bản đồ thế giới mở (Chapter 1), hệ thống dữ liệu từ vựng trong SQLite được phân mảnh (Data Partitioning) và gắn kết trực tiếp với từng loại quái vật theo cấu trúc Cây phụ thuộc (Dependency Tree). Cụ thể, kiến thức được mở khóa tuần tự dựa trên tiến trình đánh quái của người chơi:
		\begin{itemize}
			\item \textbf{Tier 1 (Ví dụ: Slime/Cấp cơ bản):} AI chỉ truy vấn và sinh câu hỏi trắc nghiệm liên quan đến từ vựng phổ thông (A1) và thì hiện tại đơn.
			\item \textbf{Tier 2 (Ví dụ: Goblin/Cấp trung bình):} Người chơi bắt buộc phải đánh bại đủ số lượng quái Tier 1 để hệ thống mở khóa dữ liệu Tier 2. Lúc này, khi chạm trán quái Tier 2, độ khó của câu hỏi sẽ tự động nâng lên vùng từ vựng chuyên sâu hơn.
			\item \textbf{Boss Encounter (Bài kiểm tra tổng hợp):} Trận đánh Boss cuối map sẽ đóng vai trò như bài test tổng hợp (Summative Assessment), trộn lẫn toàn bộ kiến thức người chơi đã học từ các loại quái trước đó.
		\end{itemize}
		\item \textbf{Giải quyết Pain Point:} Người học tiếng Anh thường bị "ngợp" trước danh sách từ vựng khổng lồ, thiếu một lộ trình học có cấu trúc đi từ nền tảng đến nâng cao. Đồng thời, trong các game học tập, người chơi có xu hướng né tránh các bài tập khó nếu hệ thống cho phép chọn tự do. 

	\end{itemize}
	\item \textbf{Feature 6 (Tính năng Kỹ thuật): Cỗ máy Sinh nội dung Bất đồng bộ (Asynchronous Pre-generation Engine)}
	\begin{itemize}
		\item \textbf{Mô tả:} Hệ thống sở hữu một module chạy ngầm (background thread) trên Godot. Thấu hiểu việc chạy mô hình ngôn ngữ Local (Ollama) trên máy tính cá nhân sẽ tốn từ 3-10 giây để trả kết quả, module này sẽ \textit{âm thầm truy vấn dữ liệu từ SQLite và gửi request cho AI ngay từ lúc người chơi đang đi bộ trên bản đồ.} Khi người chơi thực sự chạm trán quái vật, câu hỏi và đáp án đã được chuẩn bị sẵn trong bộ nhớ đệm (Cache), giúp trận đấu bắt đầu ngay lập tức với độ trễ (latency) bằng 0.
		\item \textbf{Giải quyết Pain Point:} Trải nghiệm (UX) tệ hại của các ứng dụng tích hợp AI Local là bắt người dùng phải chờ AI một thời gian lâu. Tính năng này giải quyết triệt để rào cản phần cứng yếu, giữ cho nhịp độ game (gameplay flow) diễn ra mượt mà không bị gián đoạn.
	\end{itemize}
\end{enumerate}

\section{COMPUTATIONAL THINKING FRAMEWORK}

\subsection{Decomposition (Phân rã bài toán)}
Thay vì xử lý toàn bộ khối lượng kiến thức tiếng Anh và hệ thống game trong một luồng nguyên khối, hệ thống phân rã bài toán trên 3 khía cạnh:

\subsubsection*{1. Functional Decomposition (Phân rã theo cụm chức năng logic):}
\begin{itemize}
    \item \textbf{Progression \& Gating Logic (Logic kiểm soát tiến trình):} Quản lý điều kiện mở khóa các phân vùng kiến thức khó hơn (Tier 2, Boss) dựa trên số lượng quái Tier 1 đã bị người chơi tiêu diệt.
    \item \textbf{Asynchronous AI Generation Logic (Logic sinh nội dung bất đồng bộ):} Chịu trách nhiệm ngầm truy vấn SQLite và gọi Ollama API để chuẩn bị sẵn các câu hỏi trắc nghiệm (A, B, C, D) vào hàng đợi (Queue), loại bỏ hoàn toàn độ trễ (latency) khi load trận.
    \item \textbf{Combat Resolution Logic (Logic phân giải chiến đấu):} Đánh giá đáp án trắc nghiệm người chơi chọn so với JSON của AI, từ đó quy đổi thành các lệnh trừ HP quái/người chơi và hiển thị giải thích.
\end{itemize}

\subsubsection*{2. Task Decomposition (Phân rã luồng Sinh \& Đánh giá câu hỏi trắc nghiệm):}
Để một lượt đánh (turn) diễn ra chính xác mà không gây quá tải tài nguyên tính toán, quá trình này được chia thành các bước tuần tự:
\begin{itemize}
    \item \textbf{Bước 1 (Chuẩn bị bối cảnh):} Godot trích xuất \textit{weak\_vocab} từ SQLite và \textit{biome\_context} ghép thành Prompt.
    \item \textbf{Bước 2 (Sinh dữ liệu):} Ollama nhận Prompt, tạo ra 1 câu hỏi, 1 đáp án đúng, 3 đáp án nhiễu (\textit{distractors}) và 1 lời giải thích, trả về dưới dạng chuỗi JSON.
    \item \textbf{Bước 3 (Bóc tách \& Hiển thị):} Godot parse JSON, xáo trộn (\textit{shuffle}) ngẫu nhiên vị trí 4 nút A, B, C, D hiển thị lên UI.
    \item \textbf{Bước 4 (Đánh giá \& Cập nhật):} Xác thực thao tác click của người dùng, thực thi hiệu ứng game và lưu tiến độ xuống DB.
\end{itemize}

\subsubsection*{3. Data Decomposition (Phân rã cấu trúc dữ liệu):}
Dữ liệu được chia nhỏ (\textit{Data Partitioning}) để tối ưu truy vấn trong SQLite:
\begin{itemize}
    \item \textbf{Enemy-bound Vocabulary (Dữ liệu tĩnh):} Kho từ vựng không gom chung một bảng lớn mà được gắn chặt với từng loại quái vật (Slime mang tập từ vựng A1, Goblin mang tập từ vựng A2).
    \item \textbf{Knowledge Tracing (Dữ liệu động):} Hồ sơ học tập phân rã thành các tham số nhỏ theo từng từ: \textit{encounter\_count} (số lần gặp), \textit{correct\_count} (số lần đúng) để hệ thống định lượng được Điểm thông thạo (\textit{Mastery Score}).
\end{itemize}

\subsection{Pattern Recognition (Nhận diện mẫu)}
Hệ thống sử dụng khả năng Nhận diện mẫu trên hai khía cạnh quản lý dữ liệu:
\begin{itemize}
    \item \textbf{Nhận diện mẫu lỗi học tập (Knowledge Tracing Pattern):} Dựa vào lịch sử chọn đáp án trắc nghiệm lưu trong SQLite, thuật toán quét và nhận diện các "Mẫu điểm yếu" (\textit{Pattern of weaknesses}). Những từ vựng có tỷ lệ chọn sai sẽ ép cho AI chọn làm chủ đề cho những câu hỏi tiếp theo.
    \item \textbf{Nhận diện mẫu Cấu trúc Dữ liệu AI (JSON Parsing Pattern):} Vì Local LLM đôi khi sinh ra các văn bản có định dạng bất thường, hệ thống Godot sử dụng Regex và cơ chế Try-Catch để nhận diện "Mẫu cấu trúc chuẩn". Chỉ khi phản hồi của AI khớp chính xác với mẫu \texttt{\{ "question": "", "options": [], "correct": "", "explain": "" \}} thì hệ thống mới chấp nhận nạp vào Cache, nếu không sẽ loại bỏ để tránh crash game.
\end{itemize}

\subsection{Abstraction (Trừu tượng hóa)}
Để AI (một mô hình ngôn ngữ) giao tiếp mượt mà với Game Engine (môi trường toán học/đồ họa), quá trình trừu tượng hóa tiến hành lọc bỏ các nhiễu đồ họa và số hóa ngôn ngữ:
\begin{itemize}
    \item \textbf{Trừu tượng hóa Bối cảnh Game (Context Abstraction):} LLM không cần biết tọa độ X/Y của nhân vật hay hiệu ứng âm thanh. Game trừu tượng hóa trạng thái hiện tại thành bộ tham số đơn giản truyền vào Prompt: \textit{biome} (Hang động/Rừng), \textit{enemy\_type} (Slime/Rồng), và \textit{vocabulary\_target}.
    \item \textbf{Trừu tượng hóa Đánh giá Ngôn ngữ thành Chỉ số RPG:} Sự phức tạp của việc học ngoại ngữ được trừu tượng hóa và đóng gói thành cơ chế Game cơ bản: Trả lời đúng đáp án = \textit{Hit\_Damage} (Sát thương lên quái) + Tăng \textit{Mastery\_Score}; Trả lời sai = \textit{Take\_Damage} (Bị quái đánh) + Gọi hàm \textit{Show\_Explainable\_AI}.
\end{itemize}

\subsection{Algorithm Design (Thiết kế thuật toán)}
Từ các thành phần trên, nhóm thiết kế 2 luồng thuật toán cốt lõi để đảm bảo hiệu năng và cá nhân hóa:
\begin{itemize}
    \item \textbf{Thuật toán Tải ngầm Bất đồng bộ (Asynchronous Pre-fetching Algorithm):} Chạy luồng ngầm (\textit{Background Thread}) liên tục kiểm tra trạng thái hàng đợi câu hỏi (\textit{Queue}). Nếu số lượng câu hỏi trong $\text{Queue} < 3$, thuật toán tự động bốc từ vựng điểm yếu từ SQLite, gọi API Ollama sinh sẵn JSON và nạp vào hàng đợi ngay lúc người chơi đang đi bộ khám phá Map. (Đảm bảo Zero-latency - không có độ trễ khi bước vào Combat).
    \item \textbf{Thuật toán Điều chỉnh Độ khó Động (Dynamic Difficulty Adjustment - DDA):} Thuật toán tính toán Điểm thông thạo dựa trên tỷ lệ phản hồi chính xác:
    $$\text{Mastery Score} = \frac{\text{correct\_count}}{\text{encounter\_count}}$$
    \begin{itemize}
        \item Nếu $\text{Mastery Score} < 0.5$: Tần suất từ vựng này xuất hiện tăng lên, prompt AI tạo câu hỏi trắc nghiệm có từ vựng ngữ cảnh đơn giản hơn để người chơi lấy lại căn bản.
        \item Nếu $\text{Mastery Score} > 0.8$: Thuật toán mở khóa trích xuất các từ vựng thuộc cấp độ cao hơn (Tier tiếp theo) để đẩy vào Prompt, nâng dần thách thức cho người chơi.
    \end{itemize}
\end{itemize}
